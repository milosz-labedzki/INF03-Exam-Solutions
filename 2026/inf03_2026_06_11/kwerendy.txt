SELECT drzewa.id, stanZdrowia, nazwa FROM `drzewa`,gatunki WHERE drzewa.gatunek_id = gatunki.id AND dataPosadzenia = '2008-06-06';

SELECT rodzaj, COUNT(rodzaj) AS 'ilość' FROM `gatunki` GROUP BY rodzaj;
