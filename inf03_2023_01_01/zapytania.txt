SELECT cena,pojemnosc FROM `farby` WHERE kolor = 'czerwony';

SELECT kolor,(cena/pojemnosc) AS 'cena jednostkowa' FROM `farby`;

SELECT farby.kolor,malowanie.liczba_puszek FROM `farby`,malowanie WHERE malowanie.id_farby=farby.id_farby AND id_pomieszczenia=3;
